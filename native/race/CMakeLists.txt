cmake_minimum_required(VERSION 3.25)
project(autosport_race LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/fp:strict /utf-8)
else()
  add_compile_options(-fno-fast-math -ffp-contract=off)
endif()
add_library(race_space STATIC space.cpp contact.cpp dynamics.cpp vehicle.cpp profile.cpp)
set_target_properties(race_space PROPERTIES POSITION_INDEPENDENT_CODE ON)
option(RACE_PYTHON "Build Python bindings" ON)
if(RACE_PYTHON)
  find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_race_space NO_EXTRAS bindings.cpp dynamics_bindings.cpp)
  target_link_libraries(_race_space PRIVATE race_space)
  install(TARGETS _race_space LIBRARY DESTINATION . RUNTIME DESTINATION .)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_space test_space.cpp)
  target_link_libraries(test_space PRIVATE race_space)
  add_test(NAME race-space COMMAND test_space)
  add_executable(test_contact test_contact.cpp)
  target_link_libraries(test_contact PRIVATE race_space)
  add_test(NAME race-contact COMMAND test_contact)
  add_executable(test_vehicle test_vehicle.cpp)
  target_link_libraries(test_vehicle PRIVATE race_space)
  add_test(NAME race-vehicle COMMAND test_vehicle)
endif()
